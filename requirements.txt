#
#   source .venv/bin/activate          # then: python data_logging/joystick_flight.py
#   # or call it directly:  .venv/bin/python <script>
#

#   UV=/home/andy-li/snap/code/current/.local/bin/uv
#   "$UV" venv --python 3.12 .venv

#   pupil-apriltags  tag detection (tag_hover, drone_detect)

#   dash / plotly    interactive browser dashboards (dashboard_flight.py / dashboard_synced.py)
